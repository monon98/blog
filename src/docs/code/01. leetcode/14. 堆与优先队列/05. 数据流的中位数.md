# 数据流的中位数

> 难度：困难 | 标签：设计、数据流、堆（优先队列） | [LeetCode 原题](https://leetcode.cn/problems/find-median-from-data-stream/description/)

## 题目描述

**中位数**是有序整数列表中的中间值。如果列表的大小是偶数，则没有中间值，中位数是两个中间值的平均值。

- 例如 `arr = [2,3,4]` 的中位数是 `3` 。
- 例如 `arr = [2,3]` 的中位数是 `(2 + 3) / 2 = 2.5` 。

实现 MedianFinder 类:

- `MedianFinder()` 初始化 `MedianFinder` 对象。
- `void addNum(int num)` 将数据流中的整数 `num` 添加到数据结构中。
- `double findMedian()` 返回到目前为止所有元素的中位数。与实际答案相差 `10^-5` 以内的答案将被接受。

**示例 1：**

```
输入
["MedianFinder", "addNum", "addNum", "findMedian", "addNum", "findMedian"]
[[], [1], [2], [], [3], []]
输出
[null, null, null, 1.5, null, 2.0]

解释
MedianFinder medianFinder = new MedianFinder();
medianFinder.addNum(1);    // arr = [1]
medianFinder.addNum(2);    // arr = [1, 2]
medianFinder.findMedian(); // 返回 1.5 ((1 + 2) / 2)
medianFinder.addNum(3);    // arr[1, 2, 3]
medianFinder.findMedian(); // return 2.0
```

**提示:**

- `-10^5 <= num <= 10^5`
- 在调用 `findMedian` 之前，数据结构中至少有一个元素
- 最多 `5 * 10^4` 次调用 `addNum` 和 `findMedian`

## 题解

:::: details 点击查看题解

### 思路

对顶堆：大顶堆 `lo` 存较小的一半，小顶堆 `hi` 存较大的一半，并始终保持「`lo` 的大小等于 `hi` 或比 `hi` 多 1」，于是 `lo` 的所有元素都不大于 `hi` 的所有元素，中位数只与两堆堆顶有关——`lo` 更大时取 `lo` 堆顶，否则取两堆顶的平均值。

入堆后的平衡用三步搬运：先把新元素推入 `lo`，再把 `lo` 堆顶挪到 `hi`，若此时 `hi` 比 `lo` 大再把 `hi` 堆顶挪回 `lo`。这样写不用比较新元素与当前中位数，任何长度下都能维持不变量（`lo` 恒为下中位数所在堆），偶数长度时两堆各一半、奇数长度时 `lo` 多一个。

每次 `addNum` 是常数次堆操作 O(log n)、`findMedian` 只读堆顶 O(1)，能扛住 `5 * 10^4` 次调用。常见坑是按新元素与中位数比较再分流的写法在偶数长度时容易写错边界；每次 `findMedian` 都排序返回 O(n log n) 也会超时。

```javascript [JavaScript]
class Heap {
  constructor(compare) {
    this.data = [];
    this.compare = compare;
  }
  push(val) {
    const d = this.data;
    d.push(val);
    let i = d.length - 1;
    while (i > 0) {
      const p = (i - 1) >> 1;
      if (!this.compare(d[i], d[p])) break;
      [d[i], d[p]] = [d[p], d[i]];
      i = p;
    }
  }
  pop() {
    const d = this.data;
    const top = d[0];
    const last = d.pop();
    if (d.length) {
      d[0] = last;
      let i = 0;
      for (;;) {
        let s = i;
        const l = 2 * i + 1;
        const r = l + 1;
        if (l < d.length && this.compare(d[l], d[s])) s = l;
        if (r < d.length && this.compare(d[r], d[s])) s = r;
        if (s === i) break;
        [d[s], d[i]] = [d[i], d[s]];
        i = s;
      }
    }
    return top;
  }
  peek() {
    return this.data[0];
  }
  get size() {
    return this.data.length;
  }
}

class MedianFinder {
  constructor() {
    this.lo = new Heap((a, b) => a > b); // 大顶堆：较小的一半
    this.hi = new Heap((a, b) => a < b); // 小顶堆：较大的一半
  }

  addNum(num) {
    this.lo.push(num);
    this.hi.push(this.lo.pop());
    if (this.hi.size > this.lo.size) {
      this.lo.push(this.hi.pop());
    }
  }

  findMedian() {
    if (this.lo.size > this.hi.size) {
      return this.lo.peek();
    }
    return (this.lo.peek() + this.hi.peek()) / 2;
  }
}
```

```python [Python]
import heapq


class MedianFinder:

    def __init__(self):
        self.lo: list[int] = []  # 大顶堆（取负模拟），存较小的一半
        self.hi: list[int] = []  # 小顶堆，存较大的一半

    def addNum(self, num: int) -> None:
        heapq.heappush(self.lo, -num)
        heapq.heappush(self.hi, -heapq.heappop(self.lo))
        if len(self.hi) > len(self.lo):
            heapq.heappush(self.lo, -heapq.heappop(self.hi))

    def findMedian(self) -> float:
        if len(self.lo) > len(self.hi):
            return float(-self.lo[0])
        return (-self.lo[0] + self.hi[0]) / 2
```

```rust [Rust]
use std::cell::RefCell;
use std::cmp::Reverse;
use std::collections::BinaryHeap;

struct MedianFinder {
    lo: RefCell<BinaryHeap<i32>>, // 大顶堆：较小的一半
    hi: RefCell<BinaryHeap<Reverse<i32>>>, // 小顶堆：较大的一半
}

impl MedianFinder {
    fn new() -> Self {
        MedianFinder {
            lo: RefCell::new(BinaryHeap::new()),
            hi: RefCell::new(BinaryHeap::new()),
        }
    }

    fn add_num(&self, num: i32) {
        self.lo.borrow_mut().push(num);
        let moved = self.lo.borrow_mut().pop().unwrap();
        self.hi.borrow_mut().push(Reverse(moved));
        if self.hi.borrow().len() > self.lo.borrow().len() {
            let back = self.hi.borrow_mut().pop().unwrap().0;
            self.lo.borrow_mut().push(back);
        }
    }

    fn find_median(&self) -> f64 {
        if self.lo.borrow().len() > self.hi.borrow().len() {
            self.lo.borrow().peek().copied().unwrap() as f64
        } else {
            let lo = self.lo.borrow().peek().copied().unwrap() as f64;
            let hi = self.hi.borrow().peek().copied().unwrap().0 as f64;
            (lo + hi) / 2.0
        }
    }
}
```

```go [Go]
type MedianFinder struct {
	lo []int // 大顶堆：较小的一半
	hi []int // 小顶堆：较大的一半
}

func Constructor() MedianFinder {
	return MedianFinder{}
}

func (this *MedianFinder) AddNum(num int) {
	this.lo = heapPush(this.lo, num, func(a, b int) bool { return a > b })
	var val int
	this.lo, val = heapPop(this.lo, func(a, b int) bool { return a > b })
	this.hi = heapPush(this.hi, val, func(a, b int) bool { return a < b })
	if len(this.hi) > len(this.lo) {
		this.hi, val = heapPop(this.hi, func(a, b int) bool { return a < b })
		this.lo = heapPush(this.lo, val, func(a, b int) bool { return a > b })
	}
}

func (this *MedianFinder) FindMedian() float64 {
	if len(this.lo) > len(this.hi) {
		return float64(this.lo[0])
	}
	return float64(this.lo[0]+this.hi[0]) / 2
}

func heapPush(h []int, val int, less func(a, b int) bool) []int {
	h = append(h, val)
	i := len(h) - 1
	for i > 0 {
		p := (i - 1) / 2
		if !less(h[i], h[p]) {
			break
		}
		h[i], h[p] = h[p], h[i]
		i = p
	}
	return h
}

func heapPop(h []int, less func(a, b int) bool) ([]int, int) {
	top := h[0]
	last := h[len(h)-1]
	h = h[:len(h)-1]
	if len(h) > 0 {
		h[0] = last
		i := 0
		for {
			s, l, r := i, 2*i+1, 2*i+2
			if l < len(h) && less(h[l], h[s]) {
				s = l
			}
			if r < len(h) && less(h[r], h[s]) {
				s = r
			}
			if s == i {
				break
			}
			h[s], h[i] = h[i], h[s]
			i = s
		}
	}
	return h, top
}
```

```java [Java]
class MedianFinder {

    private final PriorityQueue<Integer> lo; // 大顶堆：较小的一半
    private final PriorityQueue<Integer> hi; // 小顶堆：较大的一半

    public MedianFinder() {
        lo = new PriorityQueue<>(Collections.reverseOrder());
        hi = new PriorityQueue<>();
    }

    public void addNum(int num) {
        lo.offer(num);
        hi.offer(lo.poll());
        if (hi.size() > lo.size()) {
            lo.offer(hi.poll());
        }
    }

    public double findMedian() {
        if (lo.size() > hi.size()) {
            return lo.peek();
        }
        return (lo.peek() + hi.peek()) / 2.0;
    }
}
```

### 复杂度分析

- 时间复杂度：O(log n)，每次 addNum 至多四次堆操作；findMedian 为 O(1)。
- 空间复杂度：O(n)，两个堆合计保存全部元素。
