# 被围绕的区域

> 难度：中等 | 标签：深度优先搜索、广度优先搜索、并查集 | [LeetCode 原题](https://leetcode.cn/problems/surrounded-regions/description/)

## 题目描述

给你一个 `m x n` 的矩阵 `board` ，由若干字符 `'X'` 和 `'O'` 组成，**捕获** 所有 **被围绕的区域**：

- **连接：**一个单元格与水平或垂直方向上相邻的单元格连接。
- **区域：连接所有**`'O'` 的单元格来形成一个区域。
- **围绕：**如果一个区域中的所有 `'O'` 单元格都不在棋盘的边缘，则该区域被包围。这样的区域 **完全** 被 `'X'` 单元格包围。

通过 **原地** 将输入矩阵中的所有 `'O'` 替换为 `'X'` 来 **捕获被围绕的区域**。你不需要返回任何值。

示例 1：

**输入：**board = [['X','X','X','X'],['X','O','O','X'],['X','X','O','X'],['X','O','X','X']]

**输出：**[['X','X','X','X'],['X','X','X','X'],['X','X','X','X'],['X','O','X','X']]

**解释：**

在上图中，底部的区域没有被捕获，因为它在 board 的边缘并且不能被围绕。

示例 2：

**输入：**board = [['X']]

**输出：**[['X']]

**提示：**

- `m == board.length`
- `n == board[i].length`
- `1 <= m, n <= 200`
- `board[i][j]` 为 `'X'` 或 `'O'`

## 题解

:::: details 点击查看题解

### 思路

正难则反——「被围绕」等价于「与边界不连通」，所以先求不会被捕获的区域：从四条边界上的 `'O'` 出发做多源 BFS，把所有与边界连通的 `'O'` 标记成临时字符 `'#'`。开局就把边界上的 `'O'` 全部入队，一次 BFS 即可覆盖全部相关连通块。

扫描结束后统一替换两遍：`'#'` 恢复成 `'O'`（与边界连通、永不被捕获），剩下的 `'O'` 改成 `'X'`（真正被包围），`'X'` 保持不动。坑：不能在入队时直接改成 `'X'`，否则会把本该保留的边界区域一并吃掉；`'#'` 这个临时标记就是用来把「保留」和「捕获」拆成两趟扫描的。m、n ≤ 200，最坏 4×10⁴ 格，裸递归 DFS 有爆栈风险，因此统一用队列实现。

```javascript [JavaScript]
var solve = function(board) {
  const rows = board.length;
  const cols = board[0].length;
  const queue = [];

  const push = (i, j) => {
    if (i >= 0 && i < rows && j >= 0 && j < cols && board[i][j] === 'O') {
      board[i][j] = '#';
      queue.push([i, j]);
    }
  };

  for (let i = 0; i < rows; i++) {
    push(i, 0);
    push(i, cols - 1);
  }
  for (let j = 0; j < cols; j++) {
    push(0, j);
    push(rows - 1, j);
  }

  let head = 0;
  while (head < queue.length) {
    const [i, j] = queue[head++];
    push(i - 1, j);
    push(i + 1, j);
    push(i, j - 1);
    push(i, j + 1);
  }

  for (let i = 0; i < rows; i++) {
    for (let j = 0; j < cols; j++) {
      if (board[i][j] === 'O') board[i][j] = 'X';
      else if (board[i][j] === '#') board[i][j] = 'O';
    }
  }
};
```

```python [Python]
from collections import deque


class Solution:
    def solve(self, board: list[list[str]]) -> None:
        """
        Do not return anything, modify board in-place instead.
        """
        rows, cols = len(board), len(board[0])
        queue = deque()

        def push(i: int, j: int) -> None:
            if 0 <= i < rows and 0 <= j < cols and board[i][j] == 'O':
                board[i][j] = '#'
                queue.append((i, j))

        for i in range(rows):
            push(i, 0)
            push(i, cols - 1)
        for j in range(cols):
            push(0, j)
            push(rows - 1, j)

        while queue:
            i, j = queue.popleft()
            push(i - 1, j)
            push(i + 1, j)
            push(i, j - 1)
            push(i, j + 1)

        for i in range(rows):
            for j in range(cols):
                if board[i][j] == 'O':
                    board[i][j] = 'X'
                elif board[i][j] == '#':
                    board[i][j] = 'O'
```

```rust [Rust]
use std::collections::VecDeque;

impl Solution {
    pub fn solve(board: &mut Vec<Vec<char>>) {
        let rows = board.len();
        let cols = board[0].len();
        let mut queue: VecDeque<(usize, usize)> = VecDeque::new();

        fn push(
            board: &mut Vec<Vec<char>>,
            queue: &mut VecDeque<(usize, usize)>,
            i: isize,
            j: isize,
        ) {
            let rows = board.len() as isize;
            let cols = board[0].len() as isize;
            if i < 0 || i >= rows || j < 0 || j >= cols {
                return;
            }
            let (ui, uj) = (i as usize, j as usize);
            if board[ui][uj] == 'O' {
                board[ui][uj] = '#';
                queue.push_back((ui, uj));
            }
        }

        for i in 0..rows {
            push(board, &mut queue, i as isize, 0);
            push(board, &mut queue, i as isize, cols as isize - 1);
        }
        for j in 0..cols {
            push(board, &mut queue, 0, j as isize);
            push(board, &mut queue, rows as isize - 1, j as isize);
        }

        while let Some((i, j)) = queue.pop_front() {
            let (i, j) = (i as isize, j as isize);
            push(board, &mut queue, i - 1, j);
            push(board, &mut queue, i + 1, j);
            push(board, &mut queue, i, j - 1);
            push(board, &mut queue, i, j + 1);
        }

        for row in board.iter_mut() {
            for cell in row.iter_mut() {
                match *cell {
                    'O' => *cell = 'X',
                    '#' => *cell = 'O',
                    _ => {}
                }
            }
        }
    }
}
```

```go [Go]
func solve(board [][]byte) {
	rows, cols := len(board), len(board[0])
	type pair struct{ i, j int }
	queue := []pair{}

	push := func(i, j int) {
		if i >= 0 && i < rows && j >= 0 && j < cols && board[i][j] == 'O' {
			board[i][j] = '#'
			queue = append(queue, pair{i, j})
		}
	}

	for i := 0; i < rows; i++ {
		push(i, 0)
		push(i, cols-1)
	}
	for j := 0; j < cols; j++ {
		push(0, j)
		push(rows-1, j)
	}

	for head := 0; head < len(queue); head++ {
		i, j := queue[head].i, queue[head].j
		push(i-1, j)
		push(i+1, j)
		push(i, j-1)
		push(i, j+1)
	}

	for i := 0; i < rows; i++ {
		for j := 0; j < cols; j++ {
			switch board[i][j] {
			case 'O':
				board[i][j] = 'X'
			case '#':
				board[i][j] = 'O'
			}
		}
	}
}
```

```java [Java]
class Solution {
    private char[][] board;
    private int rows;
    private int cols;
    private Queue<int[]> queue;

    public void solve(char[][] board) {
        this.board = board;
        this.rows = board.length;
        this.cols = board[0].length;
        this.queue = new ArrayDeque<>();
        for (int i = 0; i < rows; i++) {
            push(i, 0);
            push(i, cols - 1);
        }
        for (int j = 0; j < cols; j++) {
            push(0, j);
            push(rows - 1, j);
        }
        while (!queue.isEmpty()) {
            int[] cur = queue.poll();
            int i = cur[0], j = cur[1];
            push(i - 1, j);
            push(i + 1, j);
            push(i, j - 1);
            push(i, j + 1);
        }
        for (int i = 0; i < rows; i++) {
            for (int j = 0; j < cols; j++) {
                if (board[i][j] == 'O') board[i][j] = 'X';
                else if (board[i][j] == '#') board[i][j] = 'O';
            }
        }
    }

    private void push(int i, int j) {
        if (i >= 0 && i < rows && j >= 0 && j < cols && board[i][j] == 'O') {
            board[i][j] = '#';
            queue.offer(new int[]{i, j});
        }
    }
}
```

### 复杂度分析

- 时间复杂度：O(mn)，m、n 为棋盘的行、列数，每个格子至多入队一次。
- 空间复杂度：O(mn)，最坏情况下队列同时保存 mn 个与边界连通的格子。
