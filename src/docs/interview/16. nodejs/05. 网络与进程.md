# 网络与进程

## 1. HTTP 服务器

::: details
Node.js 内置 http 模块可以创建 HTTP 服务器。

### 基本服务器

```javascript
const http = require('http');

const server = http.createServer((req, res) => {
  res.writeHead(200, { 'Content-Type': 'application/json' });
  res.end(JSON.stringify({ message: 'Hello World' }));
});

server.listen(3000, () => {
  console.log('服务器启动: http://localhost:3000');
});
```

### 路由处理

```javascript
const http = require('http');

const server = http.createServer((req, res) => {
  const { method, url } = req;
  
  // 路由处理
  if (method === 'GET' && url === '/') {
    res.end('首页');
  } else if (method === 'GET' && url === '/api/users') {
    res.setHeader('Content-Type', 'application/json');
    res.end(JSON.stringify([{ id: 1, name: '张三' }]));
  } else {
    res.statusCode = 404;
    res.end('Not Found');
  }
});
```

### 处理请求体

```javascript
const http = require('http');

const server = http.createServer((req, res) => {
  if (req.method === 'POST') {
    let body = '';
    
    req.on('data', (chunk) => {
      body += chunk.toString();
    });
    
    req.on('end', () => {
      const data = JSON.parse(body);
      console.log('收到数据:', data);
      res.end('OK');
    });
  }
});
```

### 总结

http 模块是 Node.js 网络编程的基础，实际开发推荐使用 Express 等框架。

## 2. 网络编程

::: details
Node.js 支持多种网络协议。

### TCP 服务器

```javascript
const net = require('net');

const server = net.createServer((socket) => {
  console.log('客户端连接');
  
  socket.on('data', (data) => {
    console.log('收到:', data.toString());
    socket.write('收到: ' + data);
  });
  
  socket.on('end', () => {
    console.log('客户端断开');
  });
});

server.listen(8080, () => {
  console.log('TCP 服务器启动');
});
```

### UDP 服务器

```javascript
const dgram = require('dgram');

const server = dgram.createSocket('udp4');

server.on('message', (msg, rinfo) => {
  console.log(`收到: ${msg} 来自 ${rinfo.address}:${rinfo.port}`);
});

server.bind(41234);
```

### WebSocket

```javascript
const WebSocket = require('ws');

const wss = new WebSocket.Server({ port: 8080 });

wss.on('connection', (ws) => {
  ws.on('message', (message) => {
    console.log('收到:', message);
    ws.send('回复: ' + message);
  });
});
```

### 总结

Node.js 支持 TCP、UDP、WebSocket 等多种网络协议。

## 3. 子进程

::: details
Node.js 可以创建子进程执行系统命令。

### exec

```javascript
const { exec } = require('child_process');

// 执行命令
exec('ls -la', (error, stdout, stderr) => {
  if (error) {
    console.error(error);
    return;
  }
  console.log(stdout);
});
```

### spawn

```javascript
const { spawn } = require('child_process');

// 创建子进程
const child = spawn('ls', ['-la']);

// 监听输出
child.stdout.on('data', (data) => {
  console.log(`输出: ${data}`);
});

child.stderr.on('data', (data) => {
  console.error(`错误: ${data}`);
});

child.on('close', (code) => {
  console.log(`退出码: ${code}`);
});
```

### fork

```javascript
// fork 创建 Node.js 子进程
const { fork } = require('child_process');

const child = fork('./worker.js');

child.send({ type: 'task', data: '任务数据' });

child.on('message', (msg) => {
  console.log('收到消息:', msg);
});
```

### 总结

子进程适用于执行系统命令、CPU 密集型任务等场景。

## 4. 集群

::: details
Node.js 集群可以利用多核 CPU。

### 基本集群

```javascript
const cluster = require('cluster');
const http = require('http');
const numCPUs = require('os').cpus().length;

if (cluster.isPrimary) {
  console.log(`主进程 ${process.pid}`);
  
  // 创建工作进程
  for (let i = 0; i < numCPUs; i++) {
    cluster.fork();
  }
  
  cluster.on('exit', (worker) => {
    console.log(`工作进程 ${worker.pid} 退出`);
  });
} else {
  http.createServer((req, res) => {
    res.end('处理进程: ' + process.pid);
  }).listen(3000);
  
  console.log(`工作进程 ${process.pid} 启动`);
}
```

### PM2 集群

```javascript
// ecosystem.config.js
module.exports = {
  apps: [{
    name: 'app',
    script: './app.js',
    instances: 'max',  // 使用所有 CPU
    exec_mode: 'cluster',
    env: {
      NODE_ENV: 'production'
    }
  }]
};
```

```bash
# 启动集群
pm2 start ecosystem.config.js

# 查看状态
pm2 status

# 查看日志
pm2 logs
```

### 总结

集群可以提高 Node.js 应用的并发处理能力。

## 5. 工具库

::: details
Node.js 常用的工具库。

### 错误处理

```javascript
// boom - HTTP 错误对象
const Boom = require('boom');

throw Boom.notFound('资源不存在');
throw Boom.unauthorized('未登录');
```

### 日志

```javascript
// winston - 日志库
const winston = require('winston');

const logger = winston.createLogger({
  level: 'info',
  format: winston.format.json(),
  transports: [
    new winston.transports.File({ filename: 'error.log', level: 'error' }),
    new winston.transports.File({ filename: 'combined.log' })
  ]
});

logger.info('应用启动');
logger.error('出错了');
```

### 配置管理

```javascript
// convict - 配置管理
const convict = require('convict');

const config = convict({
  env: {
    doc: '应用环境',
    format: ['production', 'development', 'test'],
    default: 'development',
    env: 'NODE_ENV'
  },
  port: {

    format: 'port',
    default: 3000,
    env: 'PORT'
  }
});
```

### 总结

合理使用工具库可以提高开发效率和代码质量。

## 6. Worker Threads

::: details
Worker Threads 用于 CPU 密集型任务，比子进程更轻量。

### 基本使用

```javascript
// main.js
const { Worker } = require('worker_threads');

const worker = new Worker('./worker.js');

worker.postMessage({ type: 'task', data: '任务数据' });

worker.on('message', (msg) => {
  console.log('收到结果:', msg);
});

worker.on('error', (err) => {
  console.error('Worker 错误:', err);
});

// worker.js
const { parentPort } = require('worker_threads');

parentPort.on('message', (msg) => {
  const result = heavyComputation(msg.data);
  parentPort.postMessage(result);
});
```

### 线程间通信

```javascript
// 共享内存
const { SharedArrayBuffer, Atomics } = require('worker_threads');

const sharedBuffer = new SharedArrayBuffer(1024);
const arr = new Int32Array(sharedBuffer);

// 主线程写入
Atomics.store(arr, 0, 42);

// Worker 线程读取
const value = Atomics.load(arr, 0);
```

### 与子进程对比

| 特性 | Worker Threads | 子进程 |
|------|----------------|--------|
| 内存 | 共享内存 | 独立内存空间 |
| 通信 | 共享内存 + postMessage | IPC 管道 |
| 开销 | 较轻 | 较重 |
| 适用场景 | CPU 密集计算 | 系统命令执行 |

## 7. 优雅关闭

::: details
优雅关闭是生产环境的必要实践。

### 实现方案

```javascript
const http = require('http');

const server = http.createServer(app);

server.listen(3000);

// 优雅关闭函数
const gracefulShutdown = (signal) => {
  console.log(`收到 ${signal} 信号，开始优雅关闭`);
  
  server.close(() => {
    console.log('HTTP 服务器已关闭');
    process.exit(0);
  });
  
  // 超时强制退出
  setTimeout(() => {
    console.error('关闭超时，强制退出');
    process.exit(1);
  }, 10000);
};

// 监听信号
process.on('SIGTERM', () => gracefulShutdown('SIGTERM'));
process.on('SIGINT', () => gracefulShutdown('SIGINT'));
```

### 数据库连接清理

```javascript
const gracefulShutdown = async () => {
  console.log('开始清理资源');
  
  // 关闭数据库连接
  await mongoose.connection.close();
  
  // 关闭 Redis 连接
  await redis.quit();
  
  // 等待现有请求完成
  server.close(() => {
    console.log('服务器已关闭');
    process.exit(0);
  });
};
```

### Docker 中的优雅关闭

```dockerfile
# 使用 exec 形式确保信号正确传递
CMD ["node", "server.js"]

# 或配合 tini
RUN apk add --no-cache tini
ENTRYPOINT ["/sbin/tini", "--"]
CMD ["node", "server.js"]
```
