# 性能优化

## 1. 启动性能优化

::: details
启动性能是用户体验的关键，优化启动速度可以提升用户满意度。

### 延迟加载

```javascript
// 不推荐：启动时加载所有模块
const { app, BrowserWindow, Menu, Tray, dialog } = require('electron');
const Store = require('electron-store');
const axios = require('axios');

// 推荐：延迟加载
app.whenReady().then(() => {
  // 只在需要时加载
  const { BrowserWindow } = require('electron');
  
  const win = new BrowserWindow({
    width: 800,
    height: 600
  });
  
  win.loadFile('index.html');
});
```

### 预热窗口

```javascript
// 预热窗口
let mainWindow;

app.whenReady().then(() => {
  // 创建但不显示
  mainWindow = new BrowserWindow({
    show: false,
    width: 800,
    height: 600
  });
  
  mainWindow.loadFile('index.html');
  
  mainWindow.once('ready-to-show', () => {
    // 准备好后再显示
    mainWindow.show();
  });
});
```

### 使用 V8 缓存

```javascript
// 启用 V8 代码缓存
app.commandLine.appendSwitch('v8-cache-options=code');

// 或在 BrowserWindow 中配置
const win = new BrowserWindow({
  webPreferences: {
    v8CacheOptions: 'code'
  }
});
```

### 减少主线程阻塞

```javascript
// 不推荐：在主线程执行耗时操作
app.whenReady().then(() => {
  // 耗时操作会阻塞主线程
  const data = readLargeFile();
  
  const win = new BrowserWindow({
    width: 800,
    height: 600
  });
  
  win.loadFile('index.html');
});

// 推荐：使用 Worker 线程
const { Worker } = require('worker_threads');

app.whenReady().then(() => {
  const worker = new Worker('./worker.js');
  
  worker.on('message', (data) => {
    // 主线程不阻塞
    const win = new BrowserWindow({
      width: 800,
      height: 600
    });
    
    win.loadFile('index.html');
  });
  
  worker.postMessage('start');
});
```

### 总结

启动性能优化包括延迟加载、预热窗口、使用 V8 缓存、减少主线程阻塞等，这些优化可以显著提升启动速度。

:::

## 2. 渲染性能优化

::: details
渲染性能直接影响用户体验，优化渲染性能可以提升应用的流畅度。

### 减少重绘和回流

```javascript
// 不推荐：频繁触发重绘
for (let i = 0; i < 1000; i++) {
  element.style.left = `${i}px`;
}

// 推荐：使用 requestAnimationFrame
function animate() {
  element.style.left = `${position}px`;
  position += 1;
  
  if (position < 1000) {
    requestAnimationFrame(animate);
  }
}
requestAnimationFrame(animate);
```

### 使用虚拟滚动

```html
<!-- 虚拟滚动 -->
<template>
  <div class="virtual-list" ref="list">
    <div class="scroll-container" :style="{ height: totalHeight + 'px' }">
      <div
        v-for="item in visibleItems"
        :key="item.id"
        class="list-item"
        :style="{ transform: `translateY(${item.position}px)` }"
      >
        {{ item.name }}
      </div>
    </div>
  </div>
</template>

<script setup>
import { ref, computed, onMounted, onUnmounted } from 'vue';

const props = defineProps({
  items: Array,
  itemHeight: Number,
  containerHeight: Number
});

const list = ref(null);
const scrollTop = ref(0);

const totalHeight = computed(() => props.items.length * props.itemHeight);

const visibleItems = computed(() => {
  const startIndex = Math.floor(scrollTop.value / props.itemHeight);
  const endIndex = Math.ceil((scrollTop.value + props.containerHeight) / props.itemHeight);
  
  return props.items.slice(startIndex, endIndex).map((item, index) => ({
    ...item,
    position: (startIndex + index) * props.itemHeight
  }));
});

function handleScroll(e) {
  scrollTop.value = e.target.scrollTop;
}

onMounted(() => {
  list.value.addEventListener('scroll', handleScroll);
});

onUnmounted(() => {
  list.value?.removeEventListener('scroll', handleScroll);
});
</script>
```

### 使用 Web Workers

```javascript
// main.js
const { ipcMain } = require('electron');

ipcMain.handle('process-data', async (event, data) => {
  // 使用 Worker 线程处理数据
  return new Promise((resolve, reject) => {
    const worker = new Worker('./worker.js');
    
    worker.on('message', (result) => {
      resolve(result);
      worker.terminate();
    });
    
    worker.on('error', (error) => {
      reject(error);
      worker.terminate();
    });
    
    worker.postMessage(data);
  });
});

// worker.js
const { parentPort } = require('worker_threads');

parentPort.on('message', (data) => {
  // 耗时的数据处理
  const result = processData(data);
  parentPort.postMessage(result);
});
```

### 使用 Canvas 替代 DOM

```html
<!-- 使用 Canvas 渲染大量数据 -->
<template>
  <canvas ref="canvas" :width="width" :height="height"></canvas>
</template>

<script setup>
import { ref, onMounted, watch } from 'vue';

const props = defineProps({
  data: Array,
  width: Number,
  height: Number
});

const canvas = ref(null);

function render() {
  const ctx = canvas.value.getContext('2d');
  ctx.clearRect(0, 0, props.width, props.height);
  
  props.data.forEach((item, index) => {
    ctx.fillStyle = item.color;
    ctx.fillRect(item.x, item.y, item.width, item.height);
  });
}

onMounted(() => {
  render();
});

watch(() => props.data, render);
</script>
```

### 总结

渲染性能优化包括减少重绘和回流、使用虚拟滚动、使用 Web Workers、使用 Canvas 替代 DOM 等，这些优化可以显著提升渲染性能。

## 3. 内存优化

::: details
内存优化可以防止内存泄漏，提升应用的稳定性。

### 检测内存泄漏

```javascript
// 检测内存使用
const memoryUsage = process.memoryUsage();
console.log(`RSS: ${memoryUsage.rss / 1024 / 1024} MB`);
console.log(`HeapUsed: ${memoryUsage.heapUsed / 1024 / 1024} MB`);
console.log(`HeapTotal: ${memoryUsage.heapTotal / 1024 / 1024} MB`);

// 定期检查
setInterval(() => {
  const memory = process.memoryUsage();
  console.log(`内存使用: ${memory.heapUsed / 1024 / 1024} MB`);
}, 60000);
```

### 释放未使用的资源

```javascript
// 清理事件监听器
class EventManager {
  constructor() {
    this.listeners = new Map();
  }
  
  on(event, callback) {
    if (!this.listeners.has(event)) {
      this.listeners.set(event, []);
    }
    this.listeners.get(event).push(callback);
  }
  
  off(event, callback) {
    if (this.listeners.has(event)) {
      const callbacks = this.listeners.get(event);
      const index = callbacks.indexOf(callback);
      if (index > -1) {
        callbacks.splice(index, 1);
      }
    }
  }
  
  clear() {
    this.listeners.clear();
  }
}

// 使用 WeakMap 避免内存泄漏
const cache = new WeakMap();

function processData(key, data) {
  if (cache.has(key)) {
    return cache.get(key);
  }
  
  const result = heavyComputation(data);
  cache.set(key, result);
  return result;
}
```

### 优化 IPC 通信

```javascript
// 不推荐：频繁发送大对象
setInterval(() => {
  ipcRenderer.send('update', largeObject);
}, 1000);

// 推荐：使用 SharedArrayBuffer
const buffer = new SharedArrayBuffer(1024);
const view = new Int32Array(buffer);

// 共享内存，避免复制

```

### 使用垃圾回收

### 监控内存使用

```javascript
// 主进程监控
const { app } = require('electron');

app.on('ready', () => {
  setInterval(() => {
    const memory = process.memoryUsage();
    
    if (memory.heapUsed > 500 * 1024 * 1024) {
      console.warn('内存使用过高:', memory.heapUsed / 1024 / 1024, 'MB');
    }
  }, 60000);

// 渲染进程监控

  setInterval(() => {
    const memory = performance.memory;
    console.log(`堆内存: ${memory.usedJSHeapSize / 1024 / 1024} MB`);
  }, 60000);
}
```

### 总结

内存优化包括检测内存泄漏、释放未使用的资源、优化 IPC 通信、使用垃圾回收、监控内存使用等，这些优化可以防止内存泄漏，提升应用的稳定性。

## 4. 网络优化

::: details
网络优化可以提升数据加载速度，改善用户体验。

### 请求合并

```javascript
// 不推荐：频繁请求
for (let i = 0; i < 100; i++) {
  fetch(`/api/item/${i}`);
}

// 推荐：批量请求
fetch('/api/items', {
  method: 'POST',
  body: JSON.stringify({ ids: Array.from({ length: 100 }, (_, i) => i) })
});
```

### 数据缓存

```javascript
// 使用内存缓存
const cache = new Map();

async function fetchData(key) {
  if (cache.has(key)) {
    return cache.get(key);
  }
  
  const data = await fetch(`/api/${key}`).then(r => r.json());
  cache.set(key, data);
  
  // 设置过期时间
  setTimeout(() => cache.delete(key), 5 * 60 * 1000);
  
  return data;
}

// 使用 electron-store 持久化缓存
const Store = require('electron-store');
const store = new Store();

async function fetchDataWithPersistence(key) {
  const cached = store.get(`cache.${key}`);
  
  if (cached && Date.now() - cached.timestamp < 5 * 60 * 1000) {
    return cached.data;
  }
  
  const data = await fetch(`/api/${key}`).then(r => r.json());
  store.set(`cache.${key}`, { data, timestamp: Date.now() });
  
  return data;
}
```

### 使用 WebSocket

```javascript
// 主进程
const { WebSocketServer } = require('ws');

const wss = new WebSocketServer({ port: 8080 });

wss.on('connection', (ws) => {
  ws.on('message', (message) => {
    // 处理消息
    ws.send(`收到: ${message}`);
  });
});

// 渲染进程
const ws = new WebSocket('ws://localhost:8080');

ws.onopen = () => {
  ws.send('Hello');
};

ws.onmessage = (event) => {
  console.log(event.data);
};
```

### 压缩数据

```javascript
// 使用 gzip 压缩
const zlib = require('zlib');

// 压缩
function compress(data) {
  return new Promise((resolve, reject) => {
    zlib.gzip(JSON.stringify(data), (err, result) => {
      if (err) reject(err);
      else resolve(result);
    });
  });
}

// 解压
function decompress(data) {
  return new Promise((resolve, reject) => {
    zlib.gunzip(data, (err, result) => {
      if (err) reject(err);
      else resolve(JSON.parse(result.toString()));
    });
  });
}
```

### 总结

网络优化包括请求合并、数据缓存、使用 WebSocket、压缩数据等，这些优化可以提升数据加载速度，改善用户体验。
