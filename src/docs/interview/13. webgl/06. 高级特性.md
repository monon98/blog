# 高级特性

## 1. 帧缓冲区

::: details
帧缓冲区用于离屏渲染和后期处理。

### 创建帧缓冲区

```javascript
// 创建帧缓冲区
const framebuffer = gl.createFramebuffer();
gl.bindFramebuffer(gl.FRAMEBUFFER, framebuffer);

// 创建颜色附件
const colorTexture = gl.createTexture();
gl.bindTexture(gl.TEXTURE_2D, colorTexture);
gl.texImage2D(gl.TEXTURE_2D, 0, gl.RGBA, width, height, 0, gl.RGBA, gl.UNSIGNED_BYTE, null);
gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MIN_FILTER, gl.LINEAR);
gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MAG_FILTER, gl.LINEAR);
gl.framebufferTexture2D(gl.FRAMEBUFFER, gl.COLOR_ATTACHMENT0, gl.TEXTURE_2D, colorTexture, 0);

// 创建深度附件
const depthBuffer = gl.createRenderbuffer();
gl.bindRenderbuffer(gl.RENDERBUFFER, depthBuffer);
gl.renderbufferStorage(gl.RENDERBUFFER, gl.DEPTH_COMPONENT16, width, height);
gl.framebufferRenderbuffer(gl.FRAMEBUFFER, gl.DEPTH_ATTACHMENT, gl.RENDERBUFFER, depthBuffer);
```

### 使用帧缓冲区

```javascript
// 渲染到帧缓冲区
gl.bindFramebuffer(gl.FRAMEBUFFER, framebuffer);
gl.viewport(0, 0, width, height);
gl.clear(gl.COLOR_BUFFER_BIT | gl.DEPTH_BUFFER_BIT);
drawScene();

// 渲染到屏幕
gl.bindFramebuffer(gl.FRAMEBUFFER, null);
gl.viewport(0, 0, canvas.width, canvas.height);
gl.clear(gl.COLOR_BUFFER_BIT | gl.DEPTH_BUFFER_BIT);
drawFullScreenQuad(colorTexture);
```

### 总结

帧缓冲区用于离屏渲染和后期处理，是实现高级效果的基础。

:::

## 2. 后期处理

::: details
后期处理可以对渲染结果进行各种图像处理效果。

### 基本后期处理

```javascript
// 渲染到帧缓冲区
gl.bindFramebuffer(gl.FRAMEBUFFER, framebuffer);
drawScene();

// 后期处理着色器
const postProcessProgram = createPostProcessProgram();

gl.bindFramebuffer(gl.FRAMEBUFFER, null);
gl.useProgram(postProcessProgram);

// 绑定渲染结果作为纹理
gl.activeTexture(gl.TEXTURE0);
gl.bindTexture(gl.TEXTURE_2D, colorTexture);
gl.uniform1i(gl.getUniformLocation(postProcessProgram, 'u_texture'), 0);

// 绘制全屏四边形
drawFullScreenQuad();
```

### 常见后期处理效果

```glsl
// 灰度效果
precision mediump float;
varying vec2 v_texCoord;
uniform sampler2D u_texture;

void main() {
  vec4 color = texture2D(u_texture, v_texCoord);
  float gray = dot(color.rgb, vec3(0.299, 0.587, 0.114));
  gl_FragColor = vec4(vec3(gray), color.a);
}

// 模糊效果
precision mediump float;
varying vec2 v_texCoord;
uniform sampler2D u_texture;
uniform vec2 u_texelSize;

void main() {
  vec4 color = vec4(0.0);
  color += texture2D(u_texture, v_texCoord + vec2(-1.0, -1.0) * u_texelSize) * 0.25;
  color += texture2D(u_texture, v_texCoord + vec2(1.0, -1.0) * u_texelSize) * 0.25;
  color += texture2D(u_texture, v_texCoord + vec2(-1.0, 1.0) * u_texelSize) * 0.25;
  color += texture2D(u_texture, v_texCoord + vec2(1.0, 1.0) * u_texelSize) * 0.25;
  gl_FragColor = color;
}
```

### 总结

后期处理可以实现灰度、模糊、锐化等各种图像处理效果。

## 3. 阴影

::: details
阴影是提升 3D 场景真实感的重要技术。

### 阴影贴图

```javascript
// 渲染阴影贴图
const shadowFramebuffer = gl.createFramebuffer();
gl.bindFramebuffer(gl.FRAMEBUFFER, shadowFramebuffer);

const shadowTexture = gl.createTexture();
gl.bindTexture(gl.TEXTURE_2D, shadowTexture);
gl.texImage2D(gl.TEXTURE_2D, 0, gl.DEPTH_COMPONENT, width, height, 0, gl.DEPTH_COMPONENT, gl.UNSIGNED_SHORT, null);
gl.framebufferTexture2D(gl.FRAMEBUFFER, gl.DEPTH_ATTACHMENT, gl.TEXTURE_2D, shadowTexture, 0);

// 从光源角度渲染
gl.useProgram(shadowProgram);
gl.uniformMatrix4fv(lightViewLocation, false, lightViewMatrix);
gl.uniformMatrix4fv(lightProjectionLocation, false, lightProjectionMatrix);
drawScene();
```

### 阴影采样

```glsl
// 阴影采样着色器
precision mediump float;
varying vec4 v_shadowCoord;
uniform sampler2D u_shadowMap;

float calculateShadow() {
  vec3 shadowCoord = v_shadowCoord.xyz / v_shadowCoord.w;
  float shadowDepth = texture2D(u_shadowMap, shadowCoord.xy).r;
  
  if (shadowCoord.z > shadowDepth + 0.001) {
    return 0.5;  // 在阴影中
  }
  return 1.0;  // 不在阴影中
}

void main() {
  float shadow = calculateShadow();
  gl_FragColor = vec4(color.rgb * shadow, color.a);
}
```

### 总结

阴影贴图是实现阴影的常用技术，需要从光源角度渲染深度信息。

## 4. 实例化渲染

::: details
实例化渲染可以高效渲染大量相同几何体。

### 基本实例化

```javascript
// WebGL2 中的实例化渲染
const instanceCount = 100;

// 创建实例化属性缓冲区
const instanceBuffer = gl.createBuffer();
gl.bindBuffer(gl.ARRAY_BUFFER, instanceBuffer);
gl.bufferData(gl.ARRAY_BUFFER, instanceData, gl.STATIC_DRAW);

// 配置实例化属性
const instanceLocation = gl.getAttribLocation(program, 'a_instance');
gl.enableVertexAttribArray(instanceLocation);
gl.vertexAttribPointer(instanceLocation, 3, gl.FLOAT, false, 0, 0);
gl.vertexAttribDivisor(instanceLocation, 1);  // 每个实例更新一次

// 实例化绘制
gl.drawArraysInstanced(gl.TRIANGLES, 0, vertexCount, instanceCount);
```

### 实例化矩阵

```javascript
// 每个实例有不同的变换矩阵
const matrices = new Float32Array(instanceCount * 16);

for (let i = 0; i < instanceCount; i++) {
  const matrix = computeMatrix(i);
  matrices.set(matrix, i * 16);
}

const matrixBuffer = gl.createBuffer();
gl.bindBuffer(gl.ARRAY_BUFFER, matrixBuffer);
gl.bufferData(gl.ARRAY_BUFFER, matrices, gl.STATIC_DRAW);

// 配置矩阵属性（4 个 vec4）
for (let i = 0; i < 4; i++) {
  const location = gl.getAttribLocation(program, `a_matrix[${i}]`);
  gl.enableVertexAttribArray(location);
  gl.vertexAttribPointer(location, 4, gl.FLOAT, false, 64, i * 16);
  gl.vertexAttribDivisor(location, 1);
}
```

### 总结

实例化渲染可以高效渲染大量相同几何体，是优化大量物体渲染的关键技术。

:::
