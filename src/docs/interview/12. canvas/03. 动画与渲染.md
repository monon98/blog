# 动画与渲染

## 1. 基本动画

::: details
Canvas 2D 动画基于 requestAnimationFrame 实现。

### 基本动画循环

```javascript
function animate() {
  requestAnimationFrame(animate);
  
  // 清除画布
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  
  // 更新位置
  x += vx;
  y += vy;
  
  // 绘制
  ctx.fillRect(x, y, 50, 50);
}

animate();
```

### 弹跳球

```javascript
let x = 100;
let y = 100;
let vx = 5;
let vy = 3;
const radius = 20;

function animate() {
  requestAnimationFrame(animate);
  
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  
  // 更新位置
  x += vx;
  y += vy;
  
  // 碰撞检测
  if (x + radius > canvas.width || x - radius < 0) {
    vx = -vx;
  }
  if (y + radius > canvas.height || y - radius < 0) {
    vy = -vy;
  }
  
  // 绘制
  ctx.beginPath();
  ctx.arc(x, y, radius, 0, Math.PI * 2);
  ctx.fillStyle = 'red';
  ctx.fill();
}

animate();
```

### 缓动动画

```javascript
// 线性插值
function lerp(start, end, t) {
  return start + (end - start) * t;
}

// 缓动函数
function easeInOut(t) {
  return t < 0.5 ? 2 * t * t : -1 + (4 - 2 * t) * t;
}

let t = 0;
function animate() {
  requestAnimationFrame(animate);
  
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  
  t += 0.01;
  if (t > 1) t = 0;
  
  const x = lerp(0, canvas.width, easeInOut(t));
  ctx.fillRect(x, 100, 50, 50);
}

animate();
```

### 总结

Canvas 2D 动画基于 requestAnimationFrame，可以实现各种动画效果。

## 2. 高级动画技术

::: details
Canvas 2D 支持各种高级动画技术。

### 粒子系统

```javascript
class Particle {
  constructor(x, y) {
    this.x = x;
    this.y = y;
    this.vx = (Math.random() - 0.5) * 4;
    this.vy = (Math.random() - 0.5) * 4;
    this.life = 1;
  }
  
  update() {
    this.x += this.vx;
    this.y += this.vy;
    this.life -= 0.02;
  }
  
  draw(ctx) {
    ctx.globalAlpha = this.life;
    ctx.fillStyle = 'red';
    ctx.fillRect(this.x, this.y, 3, 3);
  }
}

const particles = [];

function animate() {
  requestAnimationFrame(animate);
  
  ctx.fillStyle = 'rgba(0, 0, 0, 0.1)';
  ctx.fillRect(0, 0, canvas.width, canvas.height);
  
  // 添加新粒子
  if (Math.random() > 0.9) {
    particles.push(new Particle(canvas.width / 2, canvas.height / 2));
  }
  
  // 更新和绘制粒子
  for (let i = particles.length - 1; i >= 0; i--) {
    particles[i].update();
    particles[i].draw(ctx);
    
    if (particles[i].life <= 0) {
      particles.splice(i, 1);
    }
  }
  
  ctx.globalAlpha = 1;
}

animate();
```

### 帧动画

```javascript
const spriteSheet = new Image();
spriteSheet.src = 'spritesheet.png';

const frameWidth = 64;
const frameHeight = 64;
let currentFrame = 0;
const totalFrames = 8;

function animate() {
  requestAnimationFrame(animate);
  
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  
  // 绘制当前帧
  ctx.drawImage(
    spriteSheet,
    currentFrame * frameWidth, 0,
    frameWidth, frameHeight,
    100, 100,
    frameWidth, frameHeight
  );
  
  // 更新帧
  currentFrame = (currentFrame + 1) % totalFrames;
}

spriteSheet.onload = animate;
```

### 运动模糊

```javascript
function animate() {
  requestAnimationFrame(animate);
  
  // 半透明清除，产生运动模糊效果
  ctx.fillStyle = 'rgba(0, 0, 0, 0.1)';
  ctx.fillRect(0, 0, canvas.width, canvas.height);
  
  // 更新位置
  x += vx;
  
  // 绘制
  ctx.fillStyle = 'white';
  ctx.fillRect(x, 100, 50, 50);
}

animate();
```

### 总结

Canvas 2D 支持粒子系统、帧动画、运动模糊等高级动画技术。

## 3. 性能优化

::: details
Canvas 2D 动画的性能优化技巧。

### 减少重绘

```javascript
// 不推荐：每帧清除整个画布
function animate() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  // 绘制所有内容
}

// 推荐：只清除需要重绘的区域
function animate() {
  ctx.clearRect(oldX - 10, oldY - 10, 70, 70);
  ctx.fillRect(x, y, 50, 50);
  oldX = x;
  oldY = y;
}
```

### 使用离屏 Canvas

```javascript
// 创建离屏 Canvas
const offscreen = document.createElement('canvas');
offscreen.width = 100;
offscreen.height = 100;
const offCtx = offscreen.getContext('2d');

// 在离屏 Canvas 上绘制
offCtx.fillStyle = 'red';
offCtx.fillRect(0, 0, 100, 100);

// 将离屏 Canvas 绘制到主 Canvas
ctx.drawImage(offscreen, 0, 0);
```

### 批量绘制

```javascript
// 不推荐：逐个绘制
for (let i = 0; i < 1000; i++) {
  ctx.fillRect(items[i].x, items[i].y, 10, 10);
}

// 推荐：批量设置样式
ctx.fillStyle = 'red';
for (let i = 0; i < 1000; i++) {
  ctx.fillRect(items[i].x, items[i].y, 10, 10);
}
```

### 使用 requestAnimationFrame

```javascript
// 不推荐：使用 setInterval
setInterval(animate, 16);

// 推荐：使用 requestAnimationFrame
function animate() {
  requestAnimationFrame(animate);
  // 绘制逻辑
}
```

### 总结

性能优化包括减少重绘、使用离屏 Canvas、批量绘制、使用 requestAnimationFrame 等。

## 4. 帧率控制

::: details
控制动画帧率可以平衡性能和效果。

### 基本帧率控制

```javascript
let lastTime = 0;
const targetFPS = 60;
const frameInterval = 1000 / targetFPS;

function animate(currentTime) {
  requestAnimationFrame(animate);
  
  const deltaTime = currentTime - lastTime;
  
  if (deltaTime >= frameInterval) {
    lastTime = currentTime - (deltaTime % frameInterval);
    
    // 更新和绘制
    update();
    draw();
  }
}

requestAnimationFrame(animate);
```

### 帧率显示

```javascript
let frameCount = 0;
let lastTime = performance.now();

function animate() {
  requestAnimationFrame(animate);
  
  frameCount++;
  const currentTime = performance.now();
  
  if (currentTime - lastTime >= 1000) {
    console.log(`FPS: ${frameCount}`);
    frameCount = 0;
    lastTime = currentTime;
  }
  
  // 绘制逻辑
}

animate();
```

### 自适应帧率

```javascript
let targetFPS = 60;

function adjustFPS() {
  if (navigator.hardwareConcurrency) {
    const cores = navigator.hardwareConcurrency;
    targetFPS = Math.min(60, cores * 15);
  }
}

adjustFPS();
```

### 总结

帧率控制可以平衡动画的流畅度和性能消耗。
