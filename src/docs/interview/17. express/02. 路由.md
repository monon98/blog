# 路由

## 1. 路由基础

::: details
Express 的路由系统用于处理不同的 HTTP 请求。

### 基本路由

```javascript
// GET 请求
app.get('/', (req, res) => {
  res.send('首页');
});

// POST 请求
app.post('/users', (req, res) => {
  res.status(201).json({ created: true });
});

// PUT 请求
app.put('/users/:id', (req, res) => {
  res.json({ updated: true });
});

// DELETE 请求
app.delete('/users/:id', (req, res) => {
  res.json({ deleted: true });
});
```

### 路由参数

```javascript
// 动态路由参数
app.get('/users/:id', (req, res) => {
  console.log(req.params.id);
  res.json({ id: req.params.id });
});

// 多个参数
app.get('/posts/:postId/comments/:commentId', (req, res) => {
  const { postId, commentId } = req.params;
  res.json({ postId, commentId });
});
```

### 路由组

```javascript
// 使用 Router 组织路由
const router = express.Router();

router.get('/', (req, res) => {
  res.json({ users: [] });
});

router.get('/:id', (req, res) => {
  res.json({ id: req.params.id });
});

router.post('/', (req, res) => {
  res.status(201).json({ created: true });
});

// 挂载路由
app.use('/api/users', router);
```

### 总结

路由是 Express 应用的核心，组织好路由结构很重要。

:::

## 2. 路由参数

::: details
路由参数用于捕获 URL 中的动态值。

### 参数验证

```javascript
// 参数中间件
const validateId = (req, res, next) => {
  const { id } = req.params;
  if (!/^\d+$/.test(id)) {
    return res.status(400).json({ error: '无效的 ID' });
  }
  next();
};

app.get('/users/:id', validateId, (req, res) => {
  res.json({ id: req.params.id });
});
```

### 参数提取

```javascript
// express-validator
const { body, param, validationResult } = require('express-validator');

app.get('/users/:id',
  param('id').isInt().withMessage('ID 必须是数字'),
  (req, res) => {
    const errors = validationResult(req);
    if (!errors.isEmpty()) {
      return res.status(400).json({ errors: errors.array() });
    }
    res.json({ id: req.params.id });
  }
);
```

### 路由通配符

```javascript
// 通配符匹配
app.get('/files/*', (req, res) => {
  console.log(req.params[0]);
  res.send('文件路径');
});
```

### 总结

路由参数是构建 RESTful API 的基础。

:::

## 3. 路由组织

::: details
良好的路由组织结构有助于代码维护。

### 按功能模块组织

```javascript
// routes/user.js
const router = require('express').Router();
const userController = require('../controllers/user');

router.get('/', userController.getAll);
router.get('/:id', userController.getById);
router.post('/', userController.create);
router.put('/:id', userController.update);
router.delete('/:id', userController.delete);

module.exports = router;

// app.js
app.use('/api/users', require('./routes/user'));
app.use('/api/posts', require('./routes/post'));
```

### 路由前缀

```javascript
// 统一添加 API 前缀
const apiRouter = express.Router();

apiRouter.use('/users', require('./routes/user'));
apiRouter.use('/posts', require('./routes/post'));

app.use('/api', apiRouter);
```

### 路由映射表

```javascript
// routes/index.js
const routes = {
  'GET /api/users': '获取用户列表',
  'GET /api/users/:id': '获取用户详情',
  'POST /api/users': '创建用户',
  'PUT /api/users/:id': '更新用户',
  'DELETE /api/users/:id': '删除用户'
};

// 自动生成路由文档
Object.entries(routes).forEach(([route, description]) => {
  const [method, path] = route.split(' ');
  console.log(`${method} ${path} - ${description}`);
});
```

### 总结

合理的路由组织可以提高代码的可维护性。

:::

## 4. RESTful API 设计

::: details
RESTful 是一种 API 设计风格。

### 设计原则

| 方法 | 路径 | 说明 | 状态码 |
|------|------|------|--------|
| GET | /users | 获取列表 | 200 |
| GET | /users/:id | 获取详情 | 200 |
| POST | /users | 创建 | 201 |
| PUT | /users/:id | 更新 | 200 |
| DELETE | /users/:id | 删除 | 204 |

### 实际示例

```javascript
// 用户 API
router.get('/', async (req, res) => {
  const users = await User.find();
  res.json(users);
});

router.get('/:id', async (req, res) => {
  const user = await User.findById(req.params.id);
  if (!user) {
    return res.status(404).json({ error: '用户不存在' });
  }
  res.json(user);
});

router.post('/', async (req, res) => {
  const user = await User.create(req.body);
  res.status(201).json(user);
});

router.put('/:id', async (req, res) => {
  const user = await User.findByIdAndUpdate(req.params.id, req.body, {
    new: true
  });
  res.json(user);
});

router.delete('/:id', async (req, res) => {
  await User.findByIdAndDelete(req.params.id);
  res.status(204).send();
});
```

### 分页查询

```javascript
router.get('/', async (req, res) => {
  const { page = 1, limit = 10 } = req.query;
  
  const users = await User.find()
    .skip((page - 1) * limit)
    .limit(Number(limit));
  
  const total = await User.countDocuments();
  
  res.json({
    data: users,
    pagination: {
      page: Number(page),
      limit: Number(limit),
      total
    }
  });
});
```

### 总结

RESTful API 设计让 API 更清晰、一致。
