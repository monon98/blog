# 组件通信

## 1. 父子组件通信

::: details
小程序组件通信有多种方式。

### 父传子（properties）

```javascript
// 父组件
<my-component 
  title="标题" 
  count="{{count}}"
  bind:change="onChildChange"
/>

// 子组件
Component({
  properties: {
    title: {
      type: String,
      value: '默认标题'
    },
    count: {
      type: Number,
      value: 0
    }
  },
  
  observers: {
    'count': function(newVal) {
      console.log('count 变化:', newVal);
    }
  },
  
  methods: {
    handleTap() {
      this.triggerEvent('change', { value: 123 });
    }
  }
});
```

### 子传父（triggerEvent）

```javascript
// 子组件
Component({
  methods: {
    handleTap() {
      // 触发父组件事件
      this.triggerEvent('change', {
        value: this.data.value
      }, {
        bubbles: true,      // 允许冒泡
        composed: true       // 穿透组件
      });
    }
  }
});

// 父组件
<my-component bind:change="handleChange" />

Page({
  handleChange(e) {
    const { value } = e.detail;
    console.log('收到子组件数据:', value);
  }
});
```

### 总结

父子组件通信通过 properties 和 triggerEvent 实现。

:::

## 2. 跨组件通信

::: details
跨组件通信有多种方式可以选择。

### 全局事件总线

```javascript
// utils/event-bus.js（原生小程序使用 CommonJS）
function EventBus() {
  this.events = {};
}

EventBus.prototype.on = function(name, callback) {
  if (!this.events[name]) {
    this.events[name] = [];
  }
  this.events[name].push(callback);
};

EventBus.prototype.emit = function(name, data) {
  if (this.events[name]) {
    this.events[name].forEach(function(callback) {
      callback(data);
    });
  }
};

EventBus.prototype.off = function(name, callback) {
  if (this.events[name]) {
    this.events[name] = this.events[name].filter(function(cb) {
      return cb !== callback;
    });
  }
};

module.exports = new EventBus();
```

```javascript
// 使用
var eventBus = require('../../utils/event-bus');

// 组件 A 发送
eventBus.emit('update', { id: 1 });

// 组件 B 接收
eventBus.on('update', function(data) {
  console.log('收到:', data);
});
```

### 全局数据

```javascript
// app.js
App({
  globalData: {
    userInfo: null,
    token: null
  },
  
  setUserInfo(info) {
    this.globalData.userInfo = info;
  }
});

// 页面或组件中使用
const app = getApp();
const userInfo = app.globalData.userInfo;
```

### 总结

跨组件通信可以使用事件总线或全局数据。

## 3. 组件插槽

::: details
小程序组件支持插槽，可以实现更灵活的组件设计。

### 单插槽

```javascript
// 组件定义
Component({
  options: {
    multipleSlots: true
  }
});

// 组件模板
<view class="custom-dialog">
  <view class="header">
    <slot name="header"></slot>
  </view>
  <view class="content">
    <slot></slot>
  </view>
  <view class="footer">
    <slot name="footer"></slot>
  </view>
</view>
```

```html
<!-- 使用组件 -->
<custom-dialog>
  <view slot="header">标题</view>
  <view>内容</view>
  <view slot="footer">
    <button>确定</button>
  </view>
</custom-dialog>
```

### 总结

组件插槽提供了灵活的内容分发能力。

:::

## 4. 组件样式隔离

::: details
小程序组件支持样式隔离，避免样式冲突。

### 样式隔离选项

```javascript
Component({
  options: {
    addGlobalClass: true,   // 全局样式影响
    multipleSlots: true     // 多插槽
  },
  
  // 或在 JSON 配置
  // "styleIsolation": "isolated"
});
```

| 选项 | 说明 |
|------|------|
| isolated | 完全隔离（默认） |
| apply-shared | 父组件样式影响子组件 |
| shared | 父子组件样式共享 |

### 总结

样式隔离可以避免组件间样式冲突，提高组件复用性。

:::
