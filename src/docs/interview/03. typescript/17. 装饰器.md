# TypeScript 装饰器面试题

## 1. 什么是装饰器？如何使用？

::: details
装饰器是一种特殊的声明，它可以附加到类声明、方法、属性或参数上，用于修改它们的行为。

### 启用装饰器

```json
// tsconfig.json
{
  "compilerOptions": {
    "experimentalDecorators": true,
    "emitDecoratorMetadata": true
  }
}
```

### 基本语法

```typescript
// 类装饰器
function sealed(constructor: Function) {
  Object.seal(constructor);
  Object.seal(constructor.prototype);
}

@sealed
class Greeter {
  greeting: string;
  constructor(message: string) {
    this.greeting = message;
  }
  greet() {
    return "Hello, " + this.greeting;
  }
}

// 方法装饰器
function log(
  target: any,
  propertyKey: string,
  descriptor: PropertyDescriptor
) {
  const originalMethod = descriptor.value;
  descriptor.value = function (...args: any[]) {
    console.log(`Calling ${propertyKey} with args: ${JSON.stringify(args)}`);
    const result = originalMethod.apply(this, args);
    console.log(`Result: ${result}`);
    return result;
  };
}

class Calculator {
  @log
  add(a: number, b: number): number {
    return a + b;
  }
}
```

## 2. 有哪些类型的装饰器？

::: details
TypeScript 支持多种类型的装饰器，包括类装饰器、方法装饰器、属性装饰器和参数装饰器。

### 类装饰器

```typescript
// 类装饰器：接收类的构造函数
function Sealed(constructor: Function) {
  Object.seal(constructor);
  Object.seal(constructor.prototype);
}

@Sealed
class Greeter {
  greeting: string;
  constructor(message: string) {
    this.greeting = message;
  }
}

// 带参数的类装饰器工厂
function Component(selector: string) {
  return function (constructor: Function) {
    // 修改类的行为
    console.log(`Component with selector: ${selector}`);
  };
}

@Component("app-root")
class MyComponent {}
```

### 方法装饰器

```typescript
// 方法装饰器：接收目标对象、方法名和属性描述符
function Log(
  target: any,
  propertyKey: string,
  descriptor: PropertyDescriptor
) {
  const originalMethod = descriptor.value;
  descriptor.value = function (...args: any[]) {
    console.log(`Method ${propertyKey} called with args: ${JSON.stringify(args)}`);
    const result = originalMethod.apply(this, args);
    console.log(`Method ${propertyKey} returned: ${result}`);
    return result;
  };
}

class Calculator {
  @Log
  add(a: number, b: number): number {
    return a + b;
  }
}

// 只读装饰器
function ReadOnly(
  target: any,
  propertyKey: string,
  descriptor: PropertyDescriptor
) {
  descriptor.writable = false;
}

class User {
  @ReadOnly
  name: string = "Alice";
}
```

### 属性装饰器

```typescript
// 属性装饰器：接收目标对象和属性名
function Required(
  target: any,
  propertyKey: string
) {
  // 存储验证规则
  const metadataKey = `required_${propertyKey}`;
  Object.defineProperty(target, metadataKey, {
    value: true,
    enumerable: false,
    configurable: false,
  });
}

class User {
  @Required
  name!: string;

  @Required
  email!: string;
}

// 验证函数
function validate(user: User): boolean {
  const requiredFields = ["name", "email"];
  for (const field of requiredFields) {
    if (!(user as any)[field]) {
      console.log(`${field} is required`);
      return false;
    }
  }
  return true;
}
```

### 参数装饰器

```typescript
// 参数装饰器：接收目标对象、方法名和参数索引
function LogParameter(
  target: any,
  propertyKey: string,
  parameterIndex: number
) {
  const metadataKey = `log_${propertyKey}`;
  const existingParameters = Reflect.getOwnMetadata(metadataKey, target) || [];
  existingParameters.push(parameterIndex);
  Reflect.defineMetadata(metadataKey, existingParameters, target);
}

class UserService {
  createUser(
    @LogParameter name: string,
    @LogParameter email: string
  ) {
    console.log(`Creating user: ${name}, ${email}`);
  }
}
```

## 3. 装饰器的执行顺序是什么？

::: details
装饰器的执行顺序遵循特定的规则，了解这些规则可以帮助你正确使用装饰器。

### 类装饰器的执行顺序

```typescript
function First() {
  console.log("First class decorator");
  return function (constructor: Function) {
    console.log("First class decorator applied");
  };
}

function Second() {
  console.log("Second class decorator");
  return function (constructor: Function) {
    console.log("Second class decorator applied");
  };
}

@First()
@Second()
class MyClass {}

// 输出：
// First class decorator
// Second class decorator
// Second class decorator applied
// First class decorator applied
```

### 方法装饰器的执行顺序

```typescript
function Log(
  target: any,
  propertyKey: string,
  descriptor: PropertyDescriptor
) {
  console.log(`Log decorator applied to ${propertyKey}`);
}

function Cache(
  target: any,
  propertyKey: string,
  descriptor: PropertyDescriptor
) {
  console.log(`Cache decorator applied to ${propertyKey}`);
}

class UserService {
  @Log
  @Cache
  getUser(id: string) {
    return { id, name: "Alice" };
  }
}

// 输出：
// Cache decorator applied to getUser
// Log decorator applied to getUser
```

### 完整的执行顺序

```typescript
// 1. 参数装饰器（按参数索引顺序）
// 2. 方法装饰器（按方法定义顺序）
// 3. 属性装饰器（按属性定义顺序）
// 4. 类装饰器（按装饰器应用顺序）

function ParameterLog(target: any, key: string, index: number) {
  console.log(`Parameter decorator at index ${index}`);
}

function MethodLog(target: any, key: string, descriptor: PropertyDescriptor) {
  console.log(`Method decorator for ${key}`);
}

function PropertyLog(target: any, key: string) {
  console.log(`Property decorator for ${key}`);
}

function ClassLog(constructor: Function) {
  console.log("Class decorator");
}

@ClassLog
class Example {
  @PropertyLog
  name!: string;

  @MethodLog
  greet(@ParameterLog message: string): string {
    return message;
  }
}

// 输出：
// Parameter decorator at index 0
// Method decorator for greet
// Property decorator for name
// Class decorator
```

## 4. 如何使用装饰器实现依赖注入？

::: details
装饰器可以用来实现简单的依赖注入容器。

### 基本实现

```typescript
// 依赖容器
const Container = new Map<string, any>();

// 注册装饰器
function Injectable(token: string) {
  return function (constructor: Function) {
    Container.set(token, constructor);
  };
}

// 注入装饰器
function Inject(token: string) {
  return function (
    target: any,
    propertyKey: string | undefined,
    parameterIndex: number
  ) {
    const existingTokens = Reflect.getOwnMetadata("inject", target) || [];
    existingTokens.push({ token, parameterIndex });
    Reflect.defineMetadata("inject", existingTokens, target);
  };
}

// 使用示例
@Injectable("logger")
class Logger {
  log(message: string) {
    console.log(`[LOG]: ${message}`);
  }
}

@Injectable("userService")
class UserService {
  constructor(@Inject("logger") private logger: Logger) {}

  createUser(name: string) {
    this.logger.log(`Creating user: ${name}`);
    return { name };
  }
}

// 解析依赖
function resolve<T>(token: string): T {
  const constructor = Container.get(token);
  if (!constructor) {
    throw new Error(`No provider found for token: ${token}`);
  }

  const injectTokens = Reflect.getOwnMetadata("inject", constructor) || [];
  const dependencies = injectTokens
    .sort((a: any, b: any) => a.parameterIndex - b.parameterIndex)
    .map((item: any) => resolve(item.token));

  return new constructor(...dependencies);
}

// 使用
const userService = resolve<UserService>("userService");
userService.createUser("Alice");
```

## 5. 装饰器和装饰器工厂的区别是什么？

::: details
装饰器和装饰器工厂是两种不同的装饰器使用方式，它们有不同的用途和语法。

### 装饰器

```typescript
// 装饰器：直接应用到目标
function Sealed(constructor: Function) {
  Object.seal(constructor);
  Object.seal(constructor.prototype);
}

@Sealed
class Greeter {
  greeting: string;
  constructor(message: string) {
    this.greeting = message;
  }
}

// 特点：
// 1. 直接应用到目标
// 2. 不能传递参数
// 3. 语法简单
```

### 装饰器工厂

```typescript
// 装饰器工厂：返回装饰器的函数
function Component(selector: string) {
  return function (constructor: Function) {
    // 修改类的行为
    console.log(`Component with selector: ${selector}`);
  };
}

@Component("app-root")
class MyComponent {}

// 特点：
// 1. 返回装饰器的函数
// 2. 可以传递参数
// 3. 更灵活
```

### 实际应用对比

```typescript
// 1. 装饰器适合简单场景
function Log(
  target: any,
  propertyKey: string,
  descriptor: PropertyDescriptor
) {
  const originalMethod = descriptor.value;
  descriptor.value = function (...args: any[]) {
    console.log(`Calling ${propertyKey}`);
    return originalMethod.apply(this, args);
  };
}

class Calculator {
  @Log
  add(a: number, b: number): number {
    return a + b;
  }
}

// 2. 装饰器工厂适合需要参数的场景
function Throttle(delay: number) {
  return function (
    target: any,
    propertyKey: string,
    descriptor: PropertyDescriptor
  ) {
    const originalMethod = descriptor.value;
    let lastCall = 0;

    descriptor.value = function (...args: any[]) {
      const now = Date.now();
      if (now - lastCall >= delay) {
        lastCall = now;
        return originalMethod.apply(this, args);
      }
    };
  };
}

class SearchService {
  @Throttle(300)
  search(query: string) {
    console.log(`Searching: ${query}`);
  }
}
```

### 选择建议

```typescript
// 1. 使用装饰器当：
// - 不需要传递参数
// - 逻辑简单
// - 只需要基本的修改

// 2. 使用装饰器工厂当：
// - 需要传递参数
// - 逻辑复杂
// - 需要更灵活的配置
```
