# TypeScript 类型体操面试题

## 1. 请解释 TypeScript 中的工具类型（Utility Types）

::: details
TypeScript 提供了一系列内置的工具类型，用于对类型进行变换和操作。这些工具类型是 TypeScript 类型系统的核心组成部分。

### Partial\<T\>

将类型 T 的所有属性变为可选。

```typescript
interface User {
  name: string;
  age: number;
  email: string;
}

// 所有属性都变为可选
type PartialUser = Partial<User>;
// 等同于：
// {
//   name?: string;
//   age?: number;
//   email?: string;
// }

// 实际应用：更新用户信息时，所有字段都是可选的
function updateUser(id: string, updates: Partial<User>) {
  // updates 的所有属性都是可选的
}

updateUser("1", { name: "Alice" }); // ✅ 只更新 name
updateUser("1", { age: 25, email: "alice@example.com" }); // ✅ 更新多个字段
```

### Required\<T\>

将类型 T 的所有属性变为必选（与 Partial 相反）。

```typescript
interface Config {
  host?: string;
  port?: number;
  debug?: boolean;
}

// 所有属性都变为必选
type RequiredConfig = Required<Config>;
// 等同于：
// {
//   host: string;
//   port: number;
//   debug: boolean;
// }

// 实际应用：确保配置完整
function createServer(config: RequiredConfig) {
  // config 的所有属性都是必选的
}

createServer({ host: "localhost", port: 3000, debug: false }); // ✅
// createServer({ host: "localhost" }); // ❌ 缺少 port 和 debug
```

### Pick\<T, K\>

从类型 T 中选取指定的属性 K 构成新类型。

```typescript
interface User {
  name: string;
  age: number;
  email: string;
  password: string;
}

// 只选取 name 和 email
type UserBasic = Pick<User, "name" | "email">;
// 等同于：
// {
//   name: string;
//   email: string;
// }

// 实际应用：返回用户基本信息（不包含敏感字段）
function getUserInfo(id: string): UserBasic {
  return { name: "Alice", email: "alice@example.com" };
}
```

### Omit\<T, K\>

从类型 T 中移除指定的属性 K 构成新类型（与 Pick 相反）。

```typescript
interface User {
  name: string;
  age: number;
  email: string;
  password: string;
}

// 移除 password 字段
type UserWithoutPassword = Omit<User, "password">;
// 等同于：
// {
//   name: string;
//   age: number;
//   email: string;
// }

// 实际应用：创建用户时不需要 id（由数据库生成）
type CreateUserInput = Omit<User, "id">;
```

### Record\<K, V\>

构造一个属性键类型为 K，属性值类型为 V 的对象类型。

```typescript
// 基本用法
type FruitMap = Record<string, number>;
// 等同于：
// {
//   [key: string]: number;
// }

const fruits: FruitMap = {
  apple: 5,
  banana: 3,
  orange: 8,
};

// 实际应用：字典/映射类型
type UserRoles = Record<string, "admin" | "user" | "guest">;
const roles: UserRoles = {
  alice: "admin",
  bob: "user",
  charlie: "guest",
};

// 实际应用：状态管理
type LoadingState = Record<string, boolean>;
const loading: LoadingState = {
  users: false,
  posts: true,
  comments: false,
};
```

### 其他常用工具类型

```typescript
// ReturnType<T>：获取函数类型 T 的返回值类型
function getUser() {
  return { name: "Alice", age: 30 };
}
type User = ReturnType<typeof getUser>;
// { name: string; age: number }

// Parameters<T>：获取函数类型 T 的参数类型组成的元组类型
function createUser(name: string, age: number) {}
type CreateUserParams = Parameters<typeof createUser>;
// [string, number]

// Extract<T, U>：从类型 T 中提取可以赋值给类型 U 的类型
type T1 = Extract<"a" | "b" | "c", "a" | "f">;
// "a"

// Exclude<T, U>：从类型 T 中移除可以赋值给类型 U 的类型
type T2 = Exclude<"a" | "b" | "c", "a" | "f">;
// "b" | "c"

// NonNullable<T>：从类型 T 中移除 null 和 undefined
type T3 = NonNullable<string | number | null | undefined>;
// string | number

// Awaited<T>：获取 Promise<T> 的解析类型
type T4 = Awaited<Promise<string>>;
// string
```

## 2. 如何实现一个 DeepPartial 类型？

::: details
DeepPartial 是一个递归地将所有属性变为可选的类型，适用于嵌套对象。

```typescript
// 基础实现
type DeepPartial<T> = {
  [P in keyof T]?: T[P] extends object ? DeepPartial<T[P]> : T[P];
};

// 更完善的实现（处理数组和函数）
type DeepPartial<T> = T extends object
  ? {
      [P in keyof T]?: T[P] extends (infer U)[]
        ? DeepPartial<U>[]
        : T[P] extends Function
        ? T[P]
        : DeepPartial<T[P]>;
    }
  : T;

// 使用示例
interface Config {
  database: {
    host: string;
    port: number;
    credentials: {
      username: string;
      password: string;
    };
  };
  server: {
    port: number;
    cors: {
      origin: string[];
      credentials: boolean;
    };
  };
}

// 所有层级的属性都变为可选
type PartialConfig = DeepPartial<Config>;

const config: PartialConfig = {
  database: {
    host: "localhost",
    // port 和 credentials 都是可选的
  },
  // server 是可选的
};
```

## 3. 如何实现一个 DeepReadonly 类型？

::: details
DeepReadonly 是一个递归地将所有属性变为只读的类型。

```typescript
// 基础实现
type DeepReadonly<T> = {
  readonly [P in keyof T]: T[P] extends object ? DeepReadonly<T[P]> : T[P];
};

// 更完善的实现（处理数组和 Map/Set）
type DeepReadonly<T> = T extends (infer U)[]
  ? readonly DeepReadonly<U>[]
  : T extends Map<infer K, infer V>
  ? ReadonlyMap<DeepReadonly<K>, DeepReadonly<V>>
  : T extends Set<infer U>
  ? ReadonlySet<DeepReadonly<U>>
  : T extends object
  ? {
      readonly [P in keyof T]: DeepReadonly<T[P]>;
    }
  : T;

// 使用示例
interface State {
  user: {
    name: string;
    posts: {
      title: string;
      content: string;
    }[];
  };
  settings: {
    theme: string;
  };
}

const state: DeepReadonly<State> = {
  user: {
    name: "Alice",
    posts: [
      { title: "Hello", content: "World" },
    ],
  },
  settings: {
    theme: "dark",
  },
};

// state.user.name = "Bob"; // ❌ 错误：只读属性
// state.user.posts.push({ title: "New", content: "Post" }); // ❌ 错误：只读数组
```

## 4. 如何实现一个类型安全的事件发射器类型？

::: details
使用映射类型和索引访问类型实现类型安全的事件系统。

```typescript
// 事件映射类型
type EventMap = {
  login: { userId: string; timestamp: number };
  logout: { userId: string };
  error: { code: number; message: string };
};

// 类型安全的事件发射器
class TypedEventEmitter<Events extends Record<string, any>> {
  private listeners: {
    [K in keyof Events]?: Array<(data: Events[K]) => void>;
  } = {};

  on<K extends keyof Events>(event: K, listener: (data: Events[K]) => void) {
    if (!this.listeners[event]) {
      this.listeners[event] = [];
    }
    this.listeners[event]!.push(listener);
  }

  emit<K extends keyof Events>(event: K, data: Events[K]) {
    this.listeners[event]?.forEach((listener) => listener(data));
  }
}

// 使用示例
const emitter = new TypedEventEmitter<EventMap>();

emitter.on("login", (data) => {
  // data 的类型自动推断为 { userId: string; timestamp: number }
  console.log(`User ${data.userId} logged in at ${data.timestamp}`);
});

emitter.on("error", (data) => {
  // data 的类型自动推断为 { code: number; message: string }
  console.log(`Error ${data.code}: ${data.message}`);
});

emitter.emit("login", { userId: "123", timestamp: Date.now() }); // ✅
emitter.emit("error", { code: 404, message: "Not found" }); // ✅
// emitter.emit("login", { userId: "123" }); // ❌ 缺少 timestamp
// emitter.emit("unknown", {}); // ❌ 未知事件
```

## 5. 如何实现一个类型安全的路由参数类型？

::: details
使用模板字面量类型解析路由参数。

```typescript
// 解析路由参数类型
type ExtractRouteParams<T extends string> =
  T extends `${string}:${infer Param}/${infer Rest}`
    ? { [K in Param | keyof ExtractRouteParams<Rest>]: string }
    : T extends `${string}:${infer Param}`
    ? { [K in Param]: string }
    : {};

// 路由定义
type Route = "/users/:userId/posts/:postId";

// 解析结果
type RouteParams = ExtractRouteParams<Route>;
// { userId: string; postId: string }

// 类型安全的路由匹配器
function matchRoute<T extends string>(
  route: T,
  path: string
): ExtractRouteParams<T> | null {
  const routeParts = route.split("/");
  const pathParts = path.split("/");

  if (routeParts.length !== pathParts.length) {
    return null;
  }

  const params: Record<string, string> = {};

  for (let i = 0; i < routeParts.length; i++) {
    if (routeParts[i].startsWith(":")) {
      params[routeParts[i].slice(1)] = pathParts[i];
    } else if (routeParts[i] !== pathParts[i]) {
      return null;
    }
  }

  return params as ExtractRouteParams<T>;
}

// 使用示例
const params = matchRoute("/users/:userId/posts/:postId", "/users/123/posts/456");
if (params) {
  console.log(params.userId); // "123"
  console.log(params.postId); // "456"
}
```
