# TypeScript 类型守卫面试题

## 1. 什么是类型守卫？有哪些类型守卫？

::: details
类型守卫是 TypeScript 中用于在运行时检查类型的表达式，它可以帮助你在特定代码块中收窄类型。

### typeof 类型守卫

```typescript
function process(value: string | number) {
  if (typeof value === "string") {
    // 这里 value 被收窄为 string
    console.log(value.toUpperCase());
  } else {
    // 这里 value 被收窄为 number
    console.log(value.toFixed(2));
  }
}
```

### instanceof 类型守卫

```typescript
class Dog {
  bark() {
    return "Woof!";
  }
}

class Cat {
  meow() {
    return "Meow!";
  }
}

function makeSound(animal: Dog | Cat) {
  if (animal instanceof Dog) {
    // 这里 animal 被收窄为 Dog
    console.log(animal.bark());
  } else {
    // 这里 animal 被收窄为 Cat
    console.log(animal.meow());
  }
}
```

### in 类型守卫

```typescript
interface Bird {
  fly(): void;
  layEggs(): void;
}

interface Fish {
  swim(): void;
  layEggs(): void;
}

function move(animal: Bird | Fish) {
  if ("fly" in animal) {
    // 这里 animal 被收窄为 Bird
    animal.fly();
  } else {
    // 这里 animal 被收窄为 Fish
    animal.swim();
  }
}
```

### 字面量类型守卫

```typescript
type Direction = "up" | "down" | "left" | "right";

function move(direction: Direction) {
  switch (direction) {
    case "up":
      // 这里 direction 被收窄为 "up"
      break;
    case "down":
      // 这里 direction 被收窄为 "down"
      break;
    // ...
  }
}
```

## 2. 如何自定义类型守卫（类型谓词）？

::: details
你可以使用类型谓词（type predicate）来创建自定义类型守卫。

### 类型谓词语法

```typescript
function isString(value: any): value is string {
  return typeof value === "string";
}

function process(value: string | number) {
  if (isString(value)) {
    // 这里 value 被收窄为 string
    console.log(value.toUpperCase());
  } else {
    // 这里 value 被收窄为 number
    console.log(value.toFixed(2));
  }
}
```

### 实际应用示例

```typescript
// 1. 检查对象属性
interface User {
  id: string;
  name: string;
  email: string;
}

interface Admin extends User {
  role: "admin";
  permissions: string[];
}

function isAdmin(user: User | Admin): user is Admin {
  return "role" in user && user.role === "admin";
}

function greet(user: User | Admin) {
  if (isAdmin(user)) {
    console.log(`Hello admin ${user.name} with permissions: ${user.permissions}`);
  } else {
    console.log(`Hello user ${user.name}`);
  }
}

// 2. 检查数组元素
function isStringArray(value: unknown): value is string[] {
  return Array.isArray(value) && value.every((item) => typeof item === "string");
}

function processArray(value: unknown) {
  if (isStringArray(value)) {
    value.forEach((item) => console.log(item.toUpperCase()));
  }
}

// 3. 检查对象是否包含特定属性
function hasProperty<T extends string>(
  obj: unknown,
  prop: T
): obj is Record<T, unknown> {
  return typeof obj === "object" && obj !== null && prop in obj;
}

function processObject(obj: unknown) {
  if (hasProperty(obj, "name")) {
    console.log(obj.name);
  }
}
```

## 3. 什么是可辨识联合（Discriminated Unions）？

::: details
可辨识联合是一种使用共同属性（辨识符）来区分联合类型中不同类型的模式。

### 基本定义

```typescript
// 每个类型都有一个共同的属性（kind）
interface Circle {
  kind: "circle";
  radius: number;
}

interface Square {
  kind: "square";
  sideLength: number;
}

interface Triangle {
  kind: "triangle";
  base: number;
  height: number;
}

type Shape = Circle | Square | Triangle;
```

### 使用可辨识联合

```typescript
function getArea(shape: Shape): number {
  switch (shape.kind) {
    case "circle":
      // 这里 shape 被收窄为 Circle
      return Math.PI * shape.radius ** 2;
    case "square":
      // 这里 shape 被收窄为 Square
      return shape.sideLength ** 2;
    case "triangle":
      // 这里 shape 被收窄为 Triangle
      return (shape.base * shape.height) / 2;
    default:
      // 穷尽性检查
      const _exhaustive: never = shape;
      return _exhaustive;
  }
}
```

### 穷尽性检查

```typescript
function getArea(shape: Shape): number {
  switch (shape.kind) {
    case "circle":
      return Math.PI * shape.radius ** 2;
    case "square":
      return shape.sideLength ** 2;
    case "triangle":
      return (shape.base * shape.height) / 2;
    default:
      // 如果遗漏了某个 case，TypeScript 会报错
      const _exhaustive: never = shape;
      return _exhaustive;
  }
}

// 如果添加新的 Shape 类型但没有处理，TypeScript 会报错
interface Rectangle {
  kind: "rectangle";
  width: number;
  height: number;
}

// 添加 Rectangle 后，上面的 switch 会报错
// 因为 Rectangle 不能赋值给 never 类型
```

## 4. 如何使用断言函数（Assertion Functions）？

::: details
断言函数是一种特殊的函数，它在运行时抛出错误来收窄类型。

### 断言语法

```typescript
function assertIsString(value: unknown): asserts value is string {
  if (typeof value !== "string") {
    throw new Error("Expected string");
  }
}

function process(value: unknown) {
  assertIsString(value);
  // 这里 value 被收窄为 string
  console.log(value.toUpperCase());
}
```

### 实际应用示例

```typescript
// 1. 断言对象属性
function assertHasId(obj: unknown): asserts obj is { id: string } {
  if (typeof obj !== "object" || obj === null || !("id" in obj)) {
    throw new Error("Expected object with id property");
  }
}

function processObject(obj: unknown) {
  assertHasId(obj);
  console.log(obj.id);
}

// 2. 断言数组非空
function assertNonEmpty<T>(arr: T[]): asserts arr is [T, ...T[]] {
  if (arr.length === 0) {
    throw new Error("Array is empty");
  }
}

function processArray(arr: string[]) {
  assertNonEmpty(arr);
  // 这里 arr 被收窄为非空数组
  console.log(arr[0].toUpperCase());
}

// 3. 断言对象存在
function assertDefined<T>(value: T | null | undefined, name: string): asserts value is T {
  if (value === null || value === undefined) {
    throw new Error(`${name} is required`);
  }
}

function processUser(user: User | null) {
  assertDefined(user, "user");
  console.log(user.name);
}
```

## 5. 类型守卫和类型断言的区别是什么？

::: details
类型守卫和类型断言都是用来收窄类型的方式，但它们有重要的区别。

### 类型守卫（Type Guard）

```typescript
// 类型守卫在运行时检查类型
function isString(value: unknown): value is string {
  return typeof value === "string";
}

function process(value: unknown) {
  if (isString(value)) {
    // 这里 value 被收窄为 string
    // 如果 isString 返回 false，TypeScript 知道 value 不是 string
    console.log(value.toUpperCase());
  }
}
```

### 类型断言（Type Assertion）

```typescript
// 类型断言告诉 TypeScript 编译器"我比你更了解这个类型"
function process(value: unknown) {
  const str = value as string;
  // 这里 str 被断言为 string
  // 但如果 value 实际上不是 string，运行时会出错
  console.log(str.toUpperCase());
}
```

### 主要区别

```typescript
// 1. 安全性
function dangerousProcess(value: unknown) {
  const str = value as string; // 不安全，可能运行时出错
  console.log(str.toUpperCase());
}

function safeProcess(value: unknown) {
  if (typeof value === "string") {
    // 安全，TypeScript 保证这里是 string
    console.log(value.toUpperCase());
  }
}

// 2. 运行时行为
function processWithGuard(value: unknown) {
  if (typeof value === "string") {
    // 运行时检查，确保安全
    console.log(value.toUpperCase());
  }
}

function processWithAssertion(value: unknown) {
  const str = value as string; // 没有运行时检查
  console.log(str.toUpperCase()); // 可能运行时出错
}

// 3. 推荐使用类型守卫
// ❌ 不推荐
const str = value as string;

// ✅ 推荐
if (typeof value === "string") {
  console.log(value.toUpperCase());
}
```
